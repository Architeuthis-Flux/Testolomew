# The regression set: `make check`. Every line's output is checked with
# @expect against what the firmware printed for that line.

# The verbs are there.
:help
@expect "help{"
@expect ":screen:ascii"
@expect ":probe"
@expect ":key"
@expect ":ui open|menu"

# The screen, as text: the View app, nothing over it.
:screen
@expect "screen{"
@expect "app: View"
@expect "panes: -"
@expect "probe: none"
@expect "camera: fixed target 27.1 22.4 8.0"

# A simulated probe on row 14, hole 3: the tracker, the row counter and the
# LEDs all see it within a second.
:probe row 14 3
@expect "ok{probe sim"
@run 800
:screen
@expect "sim: probe"
@expect "probe: track"
@expect "row: 14 hole 3"
:leds
@expect "leds{"
@expect "cursor: tracking row 14"

# The dump path un-swaps exactly what the panel received.
:screen:verify
@expect "ok{verify"

# A screen dump, in each shape, closes its frame.
:screen:ascii
@expect "screen:ascii{"
@expect "w 80 h 60 cell 3x4 grey"
@expect "}"
:screen:dump rle 4
@expect "screen:dump{"
@expect "w 240 h 240 step 4 rle"
@expect "}"
